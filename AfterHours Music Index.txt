const MUSIC_BASE =
    "https://pub-feb120eabdf4471980ef8f1331d8baab.r2.dev/afterhoursmusicmix/";


const LIBRARY_URL =
    MUSIC_BASE + "library.json";


/* =====================================================
   ELEMENTS
===================================================== */

const landing =
    document.getElementById("landing");

const enter =
    document.getElementById("enter");

const radio =
    document.getElementById("radio");

   MUSIC VARIABLES
===================================================== */

let musicFiles = [];

let currentIndex = -1;

let history = [];

let libraryLoaded = false;


/* =====================================================
   ENTER
===================================================== */

enter.addEventListener(
    "click",
    async function() {


        landing.classList.add(
            "hidden"
        );


        cornerLogo.classList.add(
            "visible"
        );


        radio.classList.add(
            "active"
        );


        status.innerText =
            "CONNECTING";


        loading.classList.add(
            "show"
        );


        try {

            await loadLibrary();

            status.innerText =
                "LIVE";

            loading.classList.remove(
                "show"
            );

            playRandomSong();

        }

        catch(error) {

            console.error(error);

            status.innerText =
                "ERROR";

            loading.innerText =
                "MUSIC LIBRARY ERROR";

        }

    }
);


/* =====================================================
   LOAD LIBRARY
===================================================== */

async function loadLibrary() {


    if (
        libraryLoaded
    ) {

        return;

    }


    const response =
        await fetch(
            LIBRARY_URL,
            {
                cache: "no-store"
            }
        );


    if (
        !response.ok
    ) {

        throw new Error(
            "Could not load library.json"
        );

    }


    const files =
        await response.json();


    if (
        !Array.isArray(files)
    ) {

        throw new Error(
            "library.json is not an array"
        );

    }


    musicFiles =
        files.filter(
            function(file) {

                return (
                    typeof file ===
                    "string"
                );

            }
        );


    if (
        musicFiles.length === 0
    ) {

        throw new Error(
            "No music files found"
        );

    }


    libraryLoaded = true;

}


/* =====================================================
   RANDOM SONG
===================================================== */

function playRandomSong() {


    if (
        musicFiles.length === 0
    ) {

        return;

    }


    let nextIndex;


    /*
       Prevent immediate repeat.
    */

    do {

        nextIndex =
            Math.floor(
                Math.random() *
                musicFiles.length
            );

    }

    while (
        musicFiles.length > 1 &&
        nextIndex === currentIndex
    );


    if (
        currentIndex >= 0
    ) {

        history.push(
            currentIndex
        );

    }


    currentIndex =
        nextIndex;


    loadSong(
        currentIndex
    );

}


/* =====================================================
   LOAD SONG
===================================================== */

function loadSong(index) {


    const filename =
        musicFiles[index];


    if (!filename) {

        return;

    }


    /*
       Properly encode spaces,
       apostrophes and special
       characters in filenames.
    */

    const songURL =
        MUSIC_BASE +
        filename
            .split("/")
            .map(
                function(part) {

                    return encodeURIComponent(
                        part
                    );

                }
            )
            .join("/");


    /*
       Reset artwork.
    */

    artwork.src =
        "vix94logo.png";


   

    songTitle.innerText =
        cleanFilename(
            filename
        );


    songArtist.innerText =
        "VIX 94'";


    /*
       Set audio source.
    */

    audio.src =
        songURL;


    audio.load();


    /*
       Read remote metadata.
    */

    readMetadata(
        songURL
    );


    /*
       Start radio.
    */

    audio.play()
        .then(
            function() {

                playButton.innerText =
                    "PAUSE";

                artwork.classList.add(
                    "playing"
                );

            }
        )
        .catch(
            function(error) {

                console.log(
                    "Autoplay waiting:",
                    error
                );

                playButton.innerText =
                    "PLAY";

                artwork.classList.remove(
                    "playing"
                );

            }
        );

}

            if (

                    0;

                audio.play();

            }

            return;

        }


        currentIndex =
            history.pop();

    }

        }

    }

/* =====================================================
   METADATA
===================================================== */

function readMetadata(
    songURL
) {


    if (
        typeof jsmediatags ===
        "undefined"
    ) {

        return;

    }


    jsmediatags.read(
        songURL,
        {


            onSuccess:
            function(result) {


                const tags =
                    result.tags;


                /*
                   TITLE
                */

                if (
                    tags.title
                ) {

                    songTitle.innerText =
                        tags.title;

                }


                /*
                   ARTIST
                */

                if (
                    tags.artist
                ) {

                    songArtist.innerText =
                        tags.artist;

                }


                /*
                   ALBUM ART
                */

                if (
                    tags.picture
                ) {


                    const picture =
                        tags.picture;


                    let binary =
                        "";


                    for (
                        let i = 0;
                        i < picture.data.length;
                        i++
                    ) {


                        binary +=
                            String.fromCharCode(
                                picture.data[i]
                            );

                    }


                    const image =
                        "data:" +
                        picture.format +
                        ";base64," +
                        window.btoa(
                            binary
                        );


                    artwork.src =
                        image;

                }

            },


            onError:
            function(error) {

                console.log(
                    "Metadata unavailable:",
                    error
                );

            }

        }
    );

}


/* =====================================================
   CLEAN FILENAME
===================================================== */

function cleanFilename(
    filename
) {


    const name =
        filename
            .split("/")
            .pop();


    return name

        .replace(
            /\.[^/.]+$/,
            ""
        )

        .replace(
            /[_-]/g,
            " "
        );

}

                rect.left
            ) /
            rect.width;


        audio.currentTime =
            percentage *
            audio.duration;

    }
);

/* =====================================================
   TIME FORMAT
===================================================== */

function formatTime(
    seconds
) {


    if (
        !seconds ||
        isNaN(seconds)
    ) {

        return "0:00";

    }


    const minutes =
        Math.floor(
            seconds / 60
        );


    const secondsPart =
        Math.floor(
            seconds % 60
        )
        .toString()
        .padStart(
            2,
            "0"
        );


    return (
        minutes +
        ":" +
        secondsPart
    );

}
