<script>
(async()=>{
  const grid=document.getElementById("discoverGrid"), input=document.getElementById("discoverSearch");
  let tracks=[];
  try{tracks=await vix94FetchLibrary()}catch(e){grid.innerHTML='<div class="vix-empty">Music library unavailable.</div>';return}
  const render=()=>{
    const q=(input.value||"").toLowerCase().trim();
    const filtered=tracks.map(vix94ParseTrack).filter(t=>(t.title+" "+t.artist).toLowerCase().includes(q)).slice(0,60);
    grid.innerHTML=filtered.length?filtered.map(t=>`<a class="vix-card" href="${vix94TrackUrl(t.filename)}"><h2>${vix94Escape(t.title)}</h2><p>${vix94Escape(t.artist)}</p><div class="vix-meta">VIEW TRACK →</div></a>`).join(""):'<div class="vix-empty">No matching tracks found.</div>';
  };
  input.addEventListener("input",render); render();
})();
</script>
